# Multi-stage build for minimal attack surface and zero binary bloat
FROM rust:1.77-bullseye as builder
WORKDIR /usr/src/sol-price-guard

COPY Cargo.toml Cargo.lock ./
COPY src ./src
COPY tests ./tests

RUN cargo build --release

FROM debian:bullseye-slim
RUN apt-get update && apt-get install -y ca-certificates && rm -rf /var/lib/apt/lists/*
WORKDIR /app
COPY --from=builder /usr/src/sol-price-guard/target/release/sol-price-guard-core /usr/local/bin/sol-price-guard-core

ENV RUST_LOG=info
USER 10001:10001

ENTRYPOINT ["/usr/local/bin/sol-price-guard-core"]
